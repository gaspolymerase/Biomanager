{# Rendering for module-defined fields. A module's custom fields are rows in organism_module_fields; their values live in the entity's `attrs` JSON. These macros are the only place that knows how each field type turns into an input or a cell, so adding a type means touching one file. Inputs are named `attr_` so they cannot collide with fixed columns — organism_service.read_attrs() reads them back out. #} {% macro field_input(field, attrs, lines=[], usernames=[]) %} {% set value = attrs.get(field.key) %} {% set name = 'attr_' ~ field.key %} {% endmacro %} {% macro field_cell(field, attrs) %} {% set value = attrs.get(field.key) %} {% if field.field_type == 'checkbox' %} {% if value %}{% endif %} {% elif value is none or value == '' %} — {% elif field.field_type == 'mono' %} {{ value }} {% elif field.field_type == 'url' %} link {% else %} {{ value }} {% endif %} {% endmacro %} {# What a custom field sorts by: numbers as numbers, dates as ISO text, yes/no as 1 or nothing (so the ticked ones gather), choices and text as shown. data-table.js reads it from td[data-sort-for]. #} {% macro sort_value(field, attrs) -%} {%- set value = attrs.get(field.key) -%} {%- if field.field_type == 'checkbox' -%}{{ '1' if value else '' }} {%- elif value is none -%} {%- else -%}{{ value }}{%- endif -%} {%- endmacro %} {# Fields flagged show_in_table, for the list header and rows. Sortable like the built-in columns: th.dt-sortable[data-sort-key="attr_"]. #} {% macro table_headers(fields) %} {% for field in fields if field.show_in_table %} {{ icon(field_icon(field), 'dt-col-icon') }}{{ field.label }} {% endfor %} {% endmacro %} {% macro table_cells(fields, attrs) %} {% for field in fields if field.show_in_table %} {{ field_cell(field, attrs) }} {% endfor %} {% endmacro %}
